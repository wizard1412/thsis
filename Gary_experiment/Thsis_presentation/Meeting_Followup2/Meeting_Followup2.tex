\section{Metric Orthogonality Analysis}

\begin{frame}
    \frametitle{Updated Metric Set}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{7 metrics analysed:}
            \begin{itemize}
                \item MAE $\downarrow$
                \item Spearman~$\rho$ $\uparrow$
                \item Weighted~$\kappa$ $\uparrow$
                \item Adjacent Accuracy (AAcc) $\uparrow$
                \item Wasserstein Distance (WD) $\downarrow$
                \item ICC(3,1) $\uparrow$
                \item Accuracy (ACC) $\uparrow$
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Changes from last meeting:}
            \begin{itemize}
                \item Replaced Kendall~$\tau_b$ with \textbf{WD}\\
                      (more informative for distribution)
                \item Replaced AUC with \textbf{ACC}\\
                      (task is 5-class, not binary)
                \item Added \textbf{ICC} to LLM summary CSV
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Hierarchical Clustering -- 3 Groups Identified}
    \begin{columns}[T]
        \begin{column}{0.60\textwidth}
            \includegraphics[width=\linewidth]{fig4_dendrogram}
        \end{column}
        \begin{column}{0.38\textwidth}
            \textbf{3 clusters}:
            \begin{itemize}
                \item \textbf{Cluster 1}: Spearman, Weighted~$\kappa$, ICC\\
                      \textit{(ordinal agreement)}
                \item \textbf{Cluster 2}: MAE, WD, ACC\\
                      \textit{(error magnitude)}
                \item \textbf{Cluster 3}: AAcc\\
                      \textit{(most independent)}
            \end{itemize}
            \vspace{0.5em}
            $\Rightarrow$ Each cluster picks \textbf{one representative}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{PCA -- PC1 Has Near-Equal Loadings}
    \begin{center}
        \includegraphics[height=0.80\textheight]{fig5_pca}
    \end{center}
\end{frame}

\begin{frame}
    \frametitle{PCA Key Observations}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Variance explained:}
            \begin{itemize}
                \item PC1: \textbf{66.7\%} (near-equal loadings 0.34--0.45)
                \item PC2: 25.8\%
                \item 2 PCs $\geq$ 80\%, \quad 3 PCs $\geq$ 95\%
            \end{itemize}
            \vspace{0.5em}
            \textbf{Curiosity (for thesis):}\\
            PC1 loadings are nearly equal\\
            $\Rightarrow$ equal-weight composite is a natural aggregate metric\\
            $\Rightarrow$ but not used by other papers (cannot compare)
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Exceptions:}
            \begin{itemize}
                \item \textbf{AAcc}: loading only 0.21 on PC1\\
                      $\Rightarrow$ most independent metric
                \item After PC1, AAcc dominates PC2\\
                      $\Rightarrow$ confirms its uniqueness
            \end{itemize}
            \vspace{0.5em}
            \textbf{Conclusion:} 3 metrics sufficient\\
            \textbf{MAE + Weighted~$\kappa$ + AAcc}\\
            (one per cluster, most interpretable)
        \end{column}
    \end{columns}
\end{frame}

\section{Summary}

\begin{frame}
    \frametitle{Summary of Updates}
    \begin{enumerate}
        \item \textbf{Metric redundancy confirmed}\\
              3 clusters found; recommend \textbf{MAE + Weighted~$\kappa$ + AAcc}
        \vspace{0.3em}
        \item \textbf{Feature Selection Results}\\
              \texttt{period\_quartile\_range} most stable (17/19 methods);\\
              LLM-Lasso 4/6 selected features consistent with top-5
        \vspace{0.3em}
        \item \textbf{Per-recording analysis}\\
              96\% AAcc; underestimation of severe cases is the main limitation
        \vspace{0.3em}
        \item \textbf{Separating Regression from Classification}\\
              GradBoost-R Pareto-dominant among regressors;\\
              KNN-clf and ExtraTrees-C Pareto-optimal among classifiers
    \end{enumerate}
\end{frame}
