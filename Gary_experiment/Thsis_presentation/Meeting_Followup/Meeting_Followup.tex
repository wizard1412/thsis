\section{Professor's Suggestions}

\begin{frame}
    \frametitle{Key Suggestions from Meeting}
    \begin{itemize}
        \item \textbf{Metric orthogonality}
        \begin{itemize}
            \item Identify collinear metrics via pairwise correlation; keep a minimal orthogonal set
            \item Move redundant metrics to appendix
        \end{itemize}
        \item \textbf{Separate Feature Selection from Prediction}
        \begin{itemize}
            \item Mixing different FS + different classifiers = comparing apples with pears
            \item Run two controlled experiments: fix one, vary the other
        \end{itemize}
        \item \textbf{Separate Regression from Classification}
        \begin{itemize}
            \item Two distinct problem formulations; must not be mixed in the same table
        \end{itemize}
        \item \textbf{Show selected features in figures}
        \begin{itemize}
            \item Each FS method result should illustrate which features were actually chosen
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Metric Orthogonality}

\begin{frame}
    \frametitle{Metric Orthogonality -- Approach}
    \begin{itemize}
        \item Problem: too many metrics confuse readers; many carry the same information
        \item Approach: pairwise Spearman correlation + hierarchical clustering + PCA
        \item Data: LLM experiments ($n{=}12$ models, 6 metrics) and ML experiments ($n{=}104$, 4 metrics)
    \end{itemize}
    \vspace{0.3em}
    Candidate metrics: MAE, Spearman~$\rho$, Weighted~$\kappa$, Adjacent~Acc., Kendall~$\tau_b$, ICC(3,1)
\end{frame}

\begin{frame}
    \frametitle{Pairwise Spearman Correlations (LLM data, $n{=}12$)}
    \begin{center}
        \includegraphics[height=0.80\textheight]{fig1_corr_heatmap_llm}
    \end{center}
\end{frame}

\begin{frame}
    \frametitle{Hierarchical Clustering of Metrics}
    \begin{columns}[T]
        \begin{column}{0.60\textwidth}
            \includegraphics[width=\linewidth]{fig4_dendrogram}
        \end{column}
        \begin{column}{0.38\textwidth}
            \textbf{3 clusters found} (threshold $|r_s|{>}0.75$):
            \begin{itemize}
                \item \textbf{Cluster 1}: Spearman~$\rho$, Weighted~$\kappa$, Kendall~$\tau_b$, ICC(3,1)
                \item \textbf{Cluster 2}: MAE, Binary F1
                \item \textbf{Cluster 3}: Adjacent Acc.\ \textit{(most independent)}
            \end{itemize}
            \vspace{0.5em}
            \textbf{PCA}: PC1 = \textbf{78.7\%} variance\\
            $\Rightarrow$ highly redundant
        \end{column}
    \end{columns}
\end{frame}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Retained (4 orthogonal metrics):}
            \begin{itemize}
                \item \textbf{MAE $\downarrow$} --- absolute error magnitude
                \item \textbf{Spearman $\rho$ $\uparrow$} --- rank correlation; represents Cluster~1
                \item \textbf{Weighted $\kappa$ $\uparrow$} --- ordinal agreement; clinical standard
