To analyze the finger tapping performance and determine the MDS-UPDRS score, we will follow these steps:

1. **Plot the Data**: Generate plots of the thumb and index finger movements over time to visualize the tapping pattern.
2. **Calculate Tap Count**: Use the provided method to calculate the number of taps based on the peaks in the movement data.
3. **Analyze Movement Characteristics**: Examine the data for hesitations, halts, amplitude decrements, and rhythm irregularities.
4. **Assign Score Based on MDS-UPDRS Criteria**: Evaluate the performance against the scoring criteria to assign a score.

Here's the Python code to generate the plots and analyze the data:

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the CSV file
data = pd.read_csv('finger_tapping_data.csv')

# Calculate the distance between thumb and index finger
thumb_index_distance = np.sqrt((data['X_thumb'] - data['X_index'])**2 + (data['Y_thumb'] - data['Y_index'])**2)

# Plot the movement of thumb and index finger over time
plt.figure(figsize=(12, 6))
plt.subplot(121)
plt.plot(data['frame'], data['X_thumb'], label='Thumb X')
plt.plot(data['frame'], data['Y_thumb'], label='Thumb Y')
plt.title('Thumb Movement (X, Y coordinates)')
plt.xlabel('Frame')
plt.ylabel('Coordinate')
plt.legend()

plt.subplot(122)
plt.plot(data['frame'], data['X_index'], label='Index X')
plt.plot(data['frame'], data['Y_index'], label='Index Y')
plt.title('Index Finger Movement (X, Y coordinates)')
plt.xlabel('Frame')
plt.ylabel('Coordinate')
plt.legend()
plt.tight_layout()
plt.show()

# Plot the distance between thumb and index finger
plt.figure(figsize=(10, 6))
plt.plot(data['frame'], thumb_index_distance)
plt.title('Distance Between Thumb and Index Finger Over Time')
plt.xlabel('Frame')
plt.ylabel('Distance (pixels)')
plt.grid(True)
plt.show()

# Calculate tap count based on peaks in the distance data
from scipy.signal import find_peaks

# Find peaks in the distance data
peaks, _ = find_peaks(thumb_index_distance, height=np.mean(thumb_index_distance), distance=10)

tap_count = len(peaks)
print(f"Number of taps detected: {tap_count}")

# Analyze rhythm and amplitude
# Calculate time intervals between taps
time_between_taps = np.diff(data['frame'][peaks])
average_time_between_taps = np.mean(time_between_taps)

print(f"Average time between taps: {average_time_between_taps} frames")

# Look for hesitations or interruptions (large gaps between taps)
hesitation_threshold = average_time_between_taps * 2
hesitations = len([t for t in time_between_taps if t > hesitation_threshold])

print(f"Number of hesitations/interruptions: {hesitations}")

# Analyze amplitude
amplitudes = thumb_index_distance[peaks]
amplitude_decrement = np.mean(amplitudes[:len(amplitudes)//2]) - np.mean(amplitudes[len(amplitudes)//2:])
relative_amplitude_decrement = (amplitude_decrement / np.mean(amplitudes)) * 100

print(f"Relative amplitude decrement: {relative_amplitude_decrement}%")

# Assign score based on MDS-UPDRS criteria
if tap_count >= 10 and hesitations == 0 and relative_amplitude_decrement < 5:
    score = 0
elif (tap_count >= 8 and tap_count < 10) or hesitations <= 2 or relative_amplitude_decrement < 20:
    score = 1
elif (tap_count >= 6 and tap_count < 8) or (hesitations > 2 and hesitations <= 5) or relative_amplitude_decrement < 40:
    score = 2
elif tap_count < 6 or hesitations > 5 or relative_amplitude_decrement >= 40:
    score = 3

# If the subject cannot perform the task, assign the highest severity
if tap_count < 3:
    score = 4

print(f"Assigned MDS-UPDRS finger tapping score: {score}")
```

   - Minor fluctuations in peak heights suggest slight variations in tap amplitude.
