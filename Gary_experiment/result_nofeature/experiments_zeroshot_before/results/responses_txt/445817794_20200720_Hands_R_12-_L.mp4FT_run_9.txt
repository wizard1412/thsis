```python
def count_taps(data):
    # Calculate the distance between thumb and index finger
    distance = ((data['X_thumb'] - data['X_index'])**2 + (data['Y_thumb'] - data['Y_index'])**2)**0.5
    
    # Find peaks where the distance is below a threshold (e.g., 20 pixels)
    taps = []
    in_tap = False
    for i in range(len(distance)):
        if distance[i] < 20 and not in_tap:
            taps.append(i)
            in_tap = True
        elif distance[i] >= 20:
            in_tap = False
            
    return len(taps)

# Calculate the number of taps
num_taps = count_taps(data)
print(f"Number of taps: {num_taps}")
```
