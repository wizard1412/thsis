def analyze_rhythm(data):
    tap_frames = []  # List to store frame numbers where taps occur
    previous_position = None
    for i, (_, row) in enumerate(data.iterrows()):
        current_x_index = row['X_index']
        if previous_position is not None:
            # Detect tap based on movement (e.g., threshold crossing)
            if abs(current_x_index - previous_position) > 10:  # Adjust threshold as needed
                tap_frames.append(i)
        previous_position = current_x_index

    intervals = []
    for i in range(1, len(tap_frames)):
        interval = tap_frames[i] - tap_frames[i-1]
        intervals.append(interval)

    return intervals

intervals = analyze_rhythm(data)
print(f"Time intervals between taps: {intervals}")
